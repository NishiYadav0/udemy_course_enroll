# Core Telegram client — MUST be kurigram, NOT pyrogram.
#
# DO NOT change this back to `pyrogram`. The original package has been
# unmaintained since 2023 and silently stopped handling newer Telegram
# protocol layers. The failure is deceptive: login succeeds, get_chat_history()
# succeeds, every API call you make by hand succeeds — but LIVE PUSH UPDATES
# are never delivered, so the bot sits at "listening for new posts..." forever
# while posts pour into the channel.
#
# kurigram is a maintained drop-in fork and still imports as `pyrogram`,
# so no code changes are needed anywhere.
#
# This regressed once already: requirements.txt still pinned pyrogram, so a
# routine `pip install -r requirements.txt` reinstalled it over kurigram and
# the bot went deaf. If you ever see that happen again, this line is why.
kurigram

# NOTE: tgcrypto is a C-extension speed booster for Pyrogram.
# On Windows it requires Microsoft C++ Build Tools to compile.
# It is OPTIONAL — Pyrogram works perfectly without it.
# On Oracle Cloud (Linux) it compiles automatically; add it back then.
# tgcrypto

# Web scraping
requests
beautifulsoup4

# TLS fingerprint impersonation — lets us mimic Chrome's TLS handshake
# so Cloudflare Bot Management on Udemy's enrollment endpoint doesn't block us.
# Without this, /subscribed-courses/ POST returns 403 regardless of cookies.
curl_cffi>=0.7

# JS-rendering fallback for SPA coupon sites (e.g. findmycourse.in) that
# inject the real Udemy link client-side — requests+BeautifulSoup alone
# cannot see it. After `pip install -r requirements.txt`, you MUST also run:
#   playwright install chromium
#   playwright install-deps      (Linux servers only — system libs)
playwright

# Hides headless-Chrome fingerprints (navigator.webdriver etc.) so Cloudflare
# doesn't serve a challenge page during browser-based enrollment.
# utils/enroll_browser.py imports this — without it, enrollment silently
# degrades and usually fails on the Cloudflare interstitial.
playwright-stealth>=2.0

# Environment management
python-dotenv
